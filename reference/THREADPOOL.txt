FILE GENERATOR AND WRITER
=========================


import java.io.FileWriter;
import java.io.PrintWriter;

public class Ex {
    public static void main(String[] args) throws Exception {
        String[] courses = {"Mathematics", "Physics", "Chemistry", "Biology", "English"};
        
        for (int i = 0; i < 5; i++) {
            String filename = "course" + (i + 1) + ".txt";
            try (PrintWriter writer = new PrintWriter(new FileWriter(filename))) {
                for (int j = 1; j <= 1000; j++) {
                    writer.printf("%d,Student%d,%s,%.2f%n", 
                                j, j, courses[i], 10 + Math.random() * 90);
                }
            }
            System.out.println("Created: " + filename);
        }
    }
}

import java.io.BufferedReader;
import java.io.FileReader;

public class Ex2 {
    public static void main(String[] args) throws Exception {
        BufferedReader reader = new BufferedReader(new FileReader("course1.txt"));
        String line;
        
        double total = 0;
        int count = 0;
        double bestGrade = 0;
        String bestStudent = "";
        
        while ((line = reader.readLine()) != null) {
            String[] data = line.split(",");
            double grade = Double.parseDouble(data[3]);
            
            total += grade;
            count++;
            
            if (grade > bestGrade) {
                bestGrade = grade;
                bestStudent = data[1];
            }
        }
        reader.close();
        
        System.out.println("Average: " + (total/count));
        System.out.println("Best student: " + bestStudent + " : " + bestGrade );
    }
}

import java.io.BufferedReader;
import java.io.FileReader;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;

public class Ex22 {
    public static void main(String[] args) throws Exception {

        Thread[] t = new Thread[5];
        for (int i = 0; i < 5; i++) {
            int N = i +1;
            String filename = "course" + N + ".txt";   // adjust naming as needed
            t[i] = new Thread(new Task(filename));
            t[i].start();
        }

 
    }
}

class Task implements Runnable {
    private final String filename;
    
    public Task(String filename) {
        this.filename = filename;
    }
    
    public void run() {
        try (BufferedReader reader = new BufferedReader(new FileReader(filename))) {
            String line;
            
            double total = 0;
            int count = 0;
            double bestGrade = 0;
            String bestStudent = "";
            
            while ((line = reader.readLine()) != null) {
                String[] data = line.split(",");
                double grade = Double.parseDouble(data[3]);
                
                total += grade;
                count++;
                
                if (grade > bestGrade) {
                    bestGrade = grade;
                    bestStudent = data[1];
                }
            }
            
            System.out.println("=== " + filename.toUpperCase() + " ===");
            System.out.println("Thread: " + Thread.currentThread().getName());
            System.out.println("Average: " + (total/count));
            System.out.println("Best student: " + bestStudent + " : " + bestGrade);
            System.out.println(); 
            
        } catch (Exception e) {
            System.err.println("Error processing " + filename + ": " + e.getMessage());
        }
    }
}

        ExecutorService executor = Executors.newFixedThreadPool(3);
        
        for (int i = 1; i <= 5; i++) {
            String filename = "course" + i + ".txt";

import java.io.BufferedReader;
import java.io.FileReader;
import java.util.concurrent.Executors;
import java.util.concurrent.ScheduledExecutorService;
import java.util.concurrent.TimeUnit;

public class Ex3 {
    public static void main(String[] args) throws Exception {
        int cores = Runtime.getRuntime().availableProcessors();
        ScheduledExecutorService executor = Executors.newScheduledThreadPool(cores);
        
        for (int i = 1; i <= 5; i++) {
            String filename = "course" + i + ".txt";
            executor.schedule(new Task(filename), i*2, TimeUnit.SECONDS);
    }

        executor.shutdown();
    }
}
